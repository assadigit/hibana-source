    80→/** One row per linked user, JSON state, no TTL (design §7.3 — "not finite"). */
    81→async function getSession(db: Db, userId: string): Promise<BotState | null> {
    82→  const rows = await db.query<{ state: string }>('SELECT state FROM telegram_bot_sessions WHERE user_id = ?', [userId])
    83→  if (rows.length === 0) return null
    84→  try {
    85→    const s = JSON.parse(rows[0].state)
    86→    return s && typeof s === 'object' && typeof s.kind === 'string' ? (s as BotState) : null
    87→  } catch {
    88→    return null
    89→  }
    90→}
    91→async function setSession(db: Db, userId: string, state: BotState): Promise<void> {
    92→  await db.execute(
    93→    'INSERT INTO telegram_bot_sessions (user_id, state, updated_at) VALUES (?, ?, ?) ON CONFLICT(user_id) DO UPDATE SET state = excluded.state, updated_at = excluded.updated_at',
    94→    [userId, JSON.stringify(state), new Date().toISOString()],
    95→  )
    96→}
    97→async function clearSession(db: Db, userId: string): Promise<void> {
    98→  await db.execute('DELETE FROM telegram_bot_sessions WHERE user_id = ?', [userId])
    99→}

   305→/** Edit the tapped message in place; fall back to a new message if the edit is rejected (too old / unchanged). */
   306→async function render(token: string, chatId: number, messageId: number, text: string, keyboard: ReplyMarkup, parseMode: 'HTML' | undefined = 'HTML'): Promise<void> {
   307→  const r = await editMessageText(token, chatId, messageId, text, parseMode, keyboard)
   308→  if (!r.ok) {
   309→    // edit failed (message unchanged / too old) — send a fresh message
   310→    await sendTelegramMessage(token, chatId, text, parseMode, keyboard)
   311→  }
   312→}

   314→/** Render a quadrant's task-list page (used by callbacks and after add-task). */
   315→async function renderQuadrant(token: string, chatId: number, db: Db, userId: string, lang: Lang, qid: number, page: number, editMessageId?: number) {
   316→  const nameRows = await db.query<{ quadrant: number; name: string }>('SELECT quadrant, name FROM sadhana_quadrant_names WHERE user_id = ?', [userId])
   317→  const names = new Map(nameRows.map((r) => [r.quadrant, r.name]))
   318→  const offset = page * SADHANA_PAGE_SIZE
   319→  const tasks = await db.query<{ id: string; title: string; emoji: string }>(
   320→    'SELECT id, title, emoji FROM sadhana_tasks WHERE user_id = ? AND quadrant = ? AND done = 0 AND deleted_at IS NULL AND cleared_at IS NULL ORDER BY pinned DESC, position ASC, created_at DESC LIMIT ? OFFSET ?',
   321→    [userId, qid, SADHANA_PAGE_SIZE, offset],
   322→  )
   323→  const countRows = await db.query<{ n: number }>(
   324→    'SELECT COUNT(*) AS n FROM sadhana_tasks WHERE user_id = ? AND quadrant = ? AND done = 0 AND deleted_at IS NULL AND cleared_at IS NULL',
   325→    [userId, qid],
   326→  )
   327→  const total = Math.max(1, Math.ceil((countRows[0]?.n ?? 0) / SADHANA_PAGE_SIZE))
   328→  const text = taskListText(lang, qid, names, tasks, page, total)
   329→  const keyboard = taskListKeyboard(lang, qid, page, total, tasks)
   330→  if (editMessageId) await render(token, chatId, editMessageId, text, keyboard)
   331→  else await sendTelegramMessage(token, chatId, text, 'HTML', keyboard)
   332→}

   334→export function registerTelegram(app: Hono<{ Variables: { user: UserRow } }>, cfg: Config) {
   335→  // Spec §15: flood guard BEFORE the secret check (the limiter must count hostile traffic
   336→  // too; 300 req/60s per IP — same value as the CF webhook rule).
   337→  const webhookLimiter: MiddlewareHandler = async (c, next) => {
   338→    if (await hitRateLimit(cfg.db, RATE_RULES.webhook, clientIp(c))) {
   339→      return c.json({ error: 'rate_limited' }, 429)
   340→    }
   341→    return next()
   342→  }
   343→
   344→  // Public webhook (secret-token header). Telegram calls this for every message (rule 11).
   345→  // Timing-safe compare (hardening 2026-08-28): a plain !== leaks a (theoretical) early-exit
   346→  // oracle on how many leading bytes match.
   347→  app.post('/api/telegram/webhook', webhookLimiter, async (c) => {
   348→    const secret = c.req.header('X-Telegram-Bot-Api-Secret-Token')
   349→    if (!secret || !cfg.telegramSecret || !timingSafeEqualStr(secret, cfg.telegramSecret)) return c.json({ error: 'forbidden' }, 403) // rule 11
   350→    const token = cfg.telegramToken
   351→    if (!token) return c.json({ error: 'telegram_not_configured' }, 503)
   352→
   353→    const parsed = UPDATE_SCHEMA.safeParse(await c.req.json().catch(() => null))
   354→    if (!parsed.success) return c.json({ ok: true }) // ignore non-message updates
   355→
   356→    const data = parsed.data
   357→
   358→    // ---- callback_query branch (inline keyboard taps) ---------------------------
   359→    // Buttons always win and REPLACE any active await_text intent (design §8.1).
   360→    if (data.callback_query) {
   361→      const cq = data.callback_query
   362→      const chatId = cq.message.chat.id
   363→      // telegram_backup is retired (session 14: backups are on-demand, not opted-in) — the
   364→      // column stays in the schema for rollback safety but no code reads it anymore.

   742→            'SELECT id, title FROM projects WHERE user_id = ? AND deleted_at IS NULL AND LOWER(title) LIKE ? ORDER BY updated_at DESC LIMIT 10',
   743→            [userId, text.toLowerCase() + '%'],
   744→          )
   745→          await setSession(cfg.db, userId, { kind: 'connect_list', noteId }) // restore connect_list so cp:<id> resolves
   746→          if (rows.length === 0) {
   747→            await sendTelegramMessage(
   748→              token,
   749→              chatId,
   750→              t(lang, `No project matches "${esc(text)}". Try again, or tap Skip.`, `پروژه‌ای برای «${esc(text)}» پیدا نشد. دوباره بنویس یا رد شو.`),
   751→              'HTML',
   752→              kb([
   753→                [{ text: t(lang, '🔍 Search', '🔍 جستجو'), callback_data: 'psearch' }, { text: t(lang, '↩️ Skip', '↩️ رد شدن'), callback_data: 'home' }],
   754→                [{ text: t(lang, '🏠 Home', '🏠 خانه'), callback_data: 'home' }],
   755→              ]),
   756→            )
   757→          } else {
   758→            await sendTelegramMessage(token, chatId, connectListText(lang, 0, 1), 'HTML', connectListKeyboard(lang, rows, 0, 1))
   759→          }
   760→          return c.json({ ok: true })
   761→        }
   762→      }
   763→
   764→      // (4) await_list_item — append the message's lines to the open list.
   765→      if (session?.kind === 'await_list_item') {
   766→        const lines = text.split(/\r?\n+/).map((l) => l.trim()).filter(Boolean)
   767→        if (lines.length) {
   768→          const total = session.items.length + lines.length
   769→          if (total > LIST_MAX_ITEMS) {
   770→            await sendTelegramMessage(token, chatId, `List is full (${LIST_MAX_ITEMS} items max) — send /done to save it.`)
   771→            return c.json({ ok: true })
   772→          }
   773→          await setSession(cfg.db, userId, { kind: 'await_list_item', items: [...session.items, ...lines.map((l) => l.slice(0, ITEM_MAX_CHARS))] })
   774→          await sendTelegramMessage(token, chatId, `✓ ${total} item${total === 1 ? '' : 's'} so far — send more, or /done to save.`)
   775→        }
   776→        return c.json({ ok: true })
   777→      }
   778→    }
   779→
   780→    // Unknown command → help pointer; anything else is an Idea (default capture, frictionless).
   781→    if (/^\//.test(text)) {
   782→      await sendTelegramMessage(token, chatId, 'Unknown command — try /help to see what I can do.')
   783→      return c.json({ ok: true })
   784→    }
   785→    await captureIdeaReply(text, false)
   786→    return c.json({ ok: true })
   787→  })
   788→
   789→  // Any authenticated user can link their own Telegram chat (§4.10): Settings shows a
   790→  // one-time code; the user sends /start <code> to the bot. Codes expire after 1h and
   791→  // generating a new one invalidates any previous unused code for the same account.
   792→  // M9 fix (2026-09-10): changed from GET to POST — this endpoint deletes + inserts in a
   793→  // transaction (state-changing), so the CSRF middleware (which only checks POST/PUT/PATCH/
   794→  // DELETE) must apply. A GET was exempt and could be triggered by <img src> or CSRF.
   795→  app.post('/api/telegram/link-code', requireAuth(cfg), async (c) => {
   796→    const user = c.get('user')
   797→    // 12 hex chars (48 bits, hardening 2026-08-28): the code links a chat to an account —
   798→    // 8 hex chars was only ~4.3B guesses. Telegram's own flood limits make brute force
   799→    // impractical either way, but doubling the entropy costs nothing.
   800→    const code = uuid().replace(/-/g, '').slice(0, 12)
   801→    await cfg.db.transaction(async (tx) => {
   802→      tx.sql('DELETE FROM telegram_links WHERE created_at < ?', [new Date(Date.now() - LINK_CODE_TTL_MS).toISOString()]) // expired sweep
   803→      tx.sql('DELETE FROM telegram_links WHERE user_id = ?', [user.id]) // one active code per account
   804→      tx.sql('INSERT INTO telegram_links (code, user_id, created_at) VALUES (?, ?, ?)', [code, user.id, new Date().toISOString()])
   805→    })
   806→    return c.json({ ok: true, code })
   807→  })
   808→
   809→  // Authed status for Settings: is this account linked, and where is the bot?
   810→  app.get('/api/telegram/status', requireAuth(cfg), async (c) => {
   811→    const user = c.get('user')
   812→    return c.json({ ok: true, bot: TELEGRAM_BOT.handle, botUrl: TELEGRAM_BOT.url, linked: !!user.telegram_chat_id, chat_id: user.telegram_chat_id })
   813→  })
   814→
   815→  // Unlink: severs the chat mapping for this account. Captures already assigned stay —
   816→  // they are account data now (rule 1); only the delivery channel is cut.
   817→  app.delete('/api/telegram/link', requireAuth(cfg), async (c) => {
   818→    const user = c.get('user')
   819→    await cfg.db.transaction(async (tx) => {
   820→      tx.sql('UPDATE users SET telegram_chat_id = NULL WHERE id = ?', [user.id])
   821→      tx.sql('DELETE FROM telegram_links WHERE user_id = ?', [user.id])
   822→    })
   823→    return c.json({ ok: true })
   824→  })
   825→}

   827→// ---- callback_query dispatcher (inline keyboard taps) ---------------------------
   828→// Every op-code is self-contained (≤ ~41 bytes, well under Telegram's 64-byte callback_data
   829→// cap). The op-code always wins and REPLACES any active await_text intent (design §8.1).
   830→async function handleCallback(
   831→  token: string,
   832→  chatId: number,
   833→  messageId: number,
   834→  user: Pick<UserRow, 'id' | 'username' | 'telegram_paused' | 'language_pref' | 'timezone' | 'sadhana_quadrant_order' | 'role'>,
   835→  data: string,
   836→  cfg: Config,
   837→  origin: string,
   838→): Promise<void> {
   839→  const db = cfg.db
   840→  const userId = user.id
   841→  const lang = user.language_pref as Lang
   842→
   843→  // 🏠 Home / ↩️ Cancel — clear any active intent and show the home keyboard.
   844→  if (data === 'home') {
   845→    await clearSession(db, userId)
   846→    await render(token, chatId, messageId, homeText(lang), homeKeyboard(lang))
   847→    return
   848→  }
   849→
   850→  // 💡 New Idea — enter await_text:idea.
   851→  if (data === 'idea') {
   852→    await setSession(db, userId, { kind: 'await_text', intent: 'idea' })
   853→    await render(token, chatId, messageId, t(lang, '💡 Send me your idea — one message. I’ll capture it as a Spark.', '💡 ایده‌ات را بفرست — یک پیام. به‌عنوان ایده ذخیره می‌کنم.'), cancelKeyboard(lang))
   854→    return
   855→  }
   856→
   857→  // 📝 Note — enter await_text:note.
   858→  if (data === 'note') {
   859→    await setSession(db, userId, { kind: 'await_text', intent: 'note' })
   860→    await render(token, chatId, messageId, t(lang, '📝 Send me the note — I’ll add it to your dashboard.', '📝 یادداشت را بفرست — به داشبوردت اضافه می‌کنم.'), cancelKeyboard(lang))
   861→    return
   862→  }
   863→
   864→  // 📋 To-do — show the user's quadrant grid (their names + order).
   865→  if (data === 'todo') {
   866→    const nameRows = await db.query<{ quadrant: number; name: string }>('SELECT quadrant, name FROM sadhana_quadrant_names WHERE user_id = ?', [userId])
   867→    const names = new Map(nameRows.map((r) => [r.quadrant, r.name]))
   868→    await render(token, chatId, messageId, quadrantGridText(lang), quadrantKeyboard(lang, names, user.sadhana_quadrant_order))
   869→    return
   870→  }
   871→
   872→  // Quadrant open: q{1-4} (page 0) or q{1-4}p{n} (page n).
   873→  const qPageMatch = data.match(/^q([1-4])p(\d+)$/)
   874→  const qOpenMatch = data.match(/^q([1-4])$/)
   875→  if (qPageMatch || qOpenMatch) {
   876→    const qid = Number((qPageMatch ?? qOpenMatch)![1])
   877→    const page = qPageMatch ? Number(qPageMatch[2]) : 0
   878→    await renderQuadrant(token, chatId, db, userId, lang, qid, page, messageId)
   879→    return
   880→  }
   881→
   882→  // Add a task to a quadrant: q{1-4}add → await_text:sadhana_task.
   883→  const qAddMatch = data.match(/^q([1-4])add$/)
   884→  if (qAddMatch) {
   885→    const qid = Number(qAddMatch[1])
   886→    await setSession(db, userId, { kind: 'await_text', intent: 'sadhana_task', quadrant: qid })
   887→    const meta = QUADRANTS.find((q) => q.id === qid)!
   888→    const nameRows = await db.query<{ quadrant: number; name: string }>('SELECT quadrant, name FROM sadhana_quadrant_names WHERE user_id = ? AND quadrant = ?', [userId, qid])
   889→    const name = nameRows[0]?.name ?? t(lang, meta.name.en, meta.name.fa)
   890→    await render(token, chatId, messageId, t(lang, `➕ New task for <b>${esc(name)}</b> — send the title:`, `➕ کار جدید برای <b>${esc(name)}</b> — عنوان را بفرست:`), cancelKeyboard(lang))
   891→    return
   892→  }
   893→
   894→  // ✅ Mark a task done: d:<uuid>. (Approved round 1, design §9 — reversible, non-losing.)
   895→  const doneMatch = data.match(/^d:(.+)$/)
   896→  if (doneMatch) {
   897→    const taskId = doneMatch[1]
   898→    const rows = await db.query<{ id: string; title: string; emoji: string; recurring: number }>(
   899→      'SELECT id, title, emoji, recurring FROM sadhana_tasks WHERE id = ? AND user_id = ? AND done = 0 AND deleted_at IS NULL',
   900→      [taskId, userId],
   901→    )
   902→    if (rows.length === 0) {
   903→      await render(token, chatId, messageId, t(lang, 'Task not found (maybe already done).', 'کار پیدا نشد (شاید已完成 است).'), homeKeyboard(lang))
   904→      return
   905→    }
   906→    const task = rows[0]
   907→    const now = new Date().toISOString()
   908→    const today = todayIn(user.timezone)
   909→    // done=1; for recurring tasks set recur_last so resetDueRecurring can re-open on the next cycle (mirrors web app).
   910→    await db.execute('UPDATE sadhana_tasks SET done = 1, updated_at = ?, recur_last = CASE WHEN recurring = 1 THEN ? ELSE recur_last END WHERE id = ? AND user_id = ?', [now, today, taskId, userId])
   911→    await render(
   912→      token,
   913→      chatId,
   914→      messageId,
   915→      `✅ ${t(lang, 'Done', 'انجام شد')}: <b>${task.emoji || '📌'} ${esc(task.title)}</b>`,
   916→      kb([
   917→        [{ text: t(lang, '↩️ Undo', '↩️ برگردان'), callback_data: `u:${taskId}` }, { text: t(lang, '🏠 Home', '🏠 خانه'), callback_data: 'home' }],
   918→      ]),
   919→    )
   920→    return
   921→  }
   922→
   923→  // ↩️ Undo a mark-done: u:<uuid>.
   924→  const undoMatch = data.match(/^u:(.+)$/)
   925→  if (undoMatch) {
   926→    const taskId = undoMatch[1]
   927→    const rows = await db.query<{ title: string; emoji: string }>('SELECT title, emoji FROM sadhana_tasks WHERE id = ? AND user_id = ? AND done = 1', [taskId, userId])
   928→    if (rows.length === 0) {
   929→      await render(token, chatId, messageId, t(lang, 'Nothing to undo.', 'چیزی برای برگردان نیست.'), homeKeyboard(lang))
   930→      return
   931→    }
   932→    const task = rows[0]
   933→    const now = new Date().toISOString()
   934→    await db.execute('UPDATE sadhana_tasks SET done = 0, recur_last = NULL, updated_at = ? WHERE id = ? AND user_id = ?', [now, taskId, userId])
   935→    await render(
   936→      token,
   937→      chatId,
   938→      messageId,
   939→      `↩️ ${t(lang, 'Reopened', 'باز شد')}: <b>${task.emoji || '📌'} ${esc(task.title)}</b>`,
   940→      kb([[{ text: t(lang, '🏠 Home', '🏠 خانه'), callback_data: 'home' }]]),
   941→    )
   942→    return
   943→  }
   944→
   945→  // 🌐 Language picker.
   946→  if (data === 'lang') {
   947→    await render(token, chatId, messageId, langText(lang), langKeyboard(lang))
   948→    return
   949→  }
   950→  // lang:en / lang:fa → update users.language_pref (single source of truth — web locale changes too, round 2 Q1).
   951→  if (data === 'lang:en' || data === 'lang:fa') {
   952→    const newLang: Lang = data === 'lang:en' ? 'en' : 'fa'
   953→    await db.execute('UPDATE users SET language_pref = ? WHERE id = ?', [newLang, userId])
   954→    await render(token, chatId, messageId, t(newLang, '✅ Language saved.', '✅ زبان ذخیره شد.'), homeKeyboard(newLang))
   955→    return
   956→  }
   957→
   958→  // ⚙️ Settings.
   959→  if (data === 'set') {
   960→    await render(
   961→      token, chatId, messageId,
   962→      settingsText(lang, user.telegram_paused === 1, user.role === 'owner'),
   963→      settingsKeyboard(lang, user.telegram_paused === 1, user.role === 'owner'),
   964→    )
   965→    return
   966→  }
   967→  // 🗄 Send backup now (Plan B, 0044 → session 14) — owner-only ON-DEMAND action: one
   968→  // tap sends one encrypted whole-DB snapshot document to THIS chat and confirms in
   969→  // place. The old toggle (auto 4×/day sends) is gone — backups never message the chat
   970→  // on their own anymore. A member's crafted 'bak' callback is a no-op (the snapshot
   971→  // must never be deliverable to a member chat).
   972→  if (data === 'bak') {
   973→    if (user.role !== 'owner') {
   974→      // The button is hidden for members, but a crafted callback must still be a no-op.
   975→      await render(token, chatId, messageId, settingsText(lang, user.telegram_paused === 1, false), settingsKeyboard(lang, user.telegram_paused === 1, false))
   976→      return
   977→    }
   978→    const r = await sendPlanBBackupToChat(cfg, userId, String(chatId))
   979→    if (r.ok) {
   980→      await render(
   981→        token, chatId, messageId,
   982→        t(
   983→          lang,
   984→          `🗄 <b>Backup sent</b> — ${new Date().toISOString().slice(11, 16)} UTC\nsha256 <code>${r.sha256.slice(0, 16)}…</code> · restore: docs/runbook.md §2b`,
   985→          `🗄 <b>پشتیبان ارسال شد</b> — ${new Date().toISOString().slice(11, 16)} UTC\nsha256 <code>${r.sha256.slice(0, 16)}…</code> · بازیابی: docs/runbook.md §2b`,
   986→        ),
   987→        settingsKeyboard(lang, user.telegram_paused === 1, true),
   988→      )
   989→    } else {
   990→      await render(
   991→        token, chatId, messageId,
   992→        t(
   993→          lang,
   994→          `⚠️ <b>Backup failed</b>\n${r.reason ?? 'unknown error'}\n\nTry again in a moment.`,
   995→          `⚠️ <b>ارسال پشتیبان ناموفق بود</b>\n${r.reason ?? 'خطای ناشناخته'}\n\n کمی بعد دوباره امتحان کن.`,
   996→        ),
   997→        settingsKeyboard(lang, user.telegram_paused === 1, true),
   998→      )
   999→    }
  1000→    return
  1001→  }
  1002→  // ⏸ Pause / ▶️ Resume reminders (toggles users.telegram_paused).
  1003→  if (data === 'pause' || data === 'resume') {
  1004→    const pause = data === 'pause'
  1005→    await db.execute('UPDATE users SET telegram_paused = ? WHERE id = ?', [pause ? 1 : 0, userId])
  1006→    const newPaused = pause
  1007→    await render(
  1008→      token, chatId, messageId,
  1009→      settingsText(lang, newPaused, user.role === 'owner'),
  1010→      settingsKeyboard(lang, newPaused, user.role === 'owner'),
  1011→    )
  1012→    return
  1013→  }
  1014→  // 🔐 Reset password (issues a one-time reset token, delivered in-chat).
  1015→  if (data === 'reset') {
  1016→    const resetToken = await createResetToken(db, userId)
  1017→    const link = `${origin}/reset.html?token=${resetToken}`
  1018→    await render(
  1019→      token,
  1020→      chatId,
  1021→      messageId,
  1022→      t(lang, `🔐 Password reset for your Hibana account:\n\n<a href="${link}">Set a new password →</a>\n\nExpires in 1 hour.`, `🔐 بازنشانی رمز هیبانا:\n\n<a href="${link}">رمز جدید →</a>\n\nتا ۱ ساعت دیگر اعتبار دارد.`),
  1023→      kb([[{ text: t(lang, '🏠 Home', '🏠 خانه'), callback_data: 'home' }]]),
  1024→    )
  1025→    return
  1026→  }
  1027→
  1028→  // ❓ Help.
  1029→  if (data === 'help') {
  1030→    await render(token, chatId, messageId, helpText(lang), helpKeyboard(lang), undefined)
  1031→    return
  1032→  }
  1033→
  1034→  // 🔗 Connect (open the project list for the note pending in session.connect_list).
  1035→  if (data === 'connect') {
  1036→    const session = await getSession(db, userId)
  1037→    if (!session || session.kind !== 'connect_list') {
  1038→      await render(token, chatId, messageId, t(lang, 'No note to connect. Tap 📝 Note first.', 'یادداشتی برای اتصال نیست. اول 📝 یادداشت را بزن.'), homeKeyboard(lang))
  1039→      return
  1040→    }
  1041→    const offset = 0
  1042→    const projects = await db.query<{ id: string; title: string }>(
  1043→      'SELECT id, title FROM projects WHERE user_id = ? AND deleted_at IS NULL ORDER BY updated_at DESC LIMIT ? OFFSET ?',
  1044→      [userId, PROJECT_PAGE_SIZE, offset],
  1045→    )
  1046→    const countRows = await db.query<{ n: number }>('SELECT COUNT(*) AS n FROM projects WHERE user_id = ? AND deleted_at IS NULL', [userId])
  1047→    const total = Math.max(1, Math.ceil((countRows[0]?.n ?? 0) / PROJECT_PAGE_SIZE))
  1048→    await render(token, chatId, messageId, connectListText(lang, 0, total), connectListKeyboard(lang, projects, 0, total))
  1049→    return
  1050→  }
  1051→  // Connect-list pagination: connectp{n}.
  1052→  const connectPageMatch = data.match(/^connectp(\d+)$/)
  1053→  if (connectPageMatch) {
  1054→    const page = Number(connectPageMatch[1])
  1055→    const session = await getSession(db, userId)
  1056→    if (!session || session.kind !== 'connect_list') {
  1057→      await render(token, chatId, messageId, homeText(lang), homeKeyboard(lang))
  1058→      return
  1059→    }
  1060→    const offset = page * PROJECT_PAGE_SIZE
  1061→    const projects = await db.query<{ id: string; title: string }>(
  1062→      'SELECT id, title FROM projects WHERE user_id = ? AND deleted_at IS NULL ORDER BY updated_at DESC LIMIT ? OFFSET ?',
  1063→      [userId, PROJECT_PAGE_SIZE, offset],
  1064→    )
  1065→    const countRows = await db.query<{ n: number }>('SELECT COUNT(*) AS n FROM projects WHERE user_id = ? AND deleted_at IS NULL', [userId])
  1066→    const total = Math.max(1, Math.ceil((countRows[0]?.n ?? 0) / PROJECT_PAGE_SIZE))
  1067→    await render(token, chatId, messageId, connectListText(lang, page, total), connectListKeyboard(lang, projects, page, total))
  1068→    return
  1069→  }
  1070→  // cp:<projectId> — connect the pending note (session.connect_list.noteId) to this project.
  1071→  const connectMatch = data.match(/^cp:(.+)$/)
  1072→  if (connectMatch) {
  1073→    const projectId = connectMatch[1]
  1074→    const session = await getSession(db, userId)
  1075→    if (!session || session.kind !== 'connect_list') {
  1076→      await render(token, chatId, messageId, t(lang, 'No note to connect.', 'یادداشتی برای اتصال نیست.'), homeKeyboard(lang))
  1077→      return
  1078→    }
  1079→    const projRows = await db.query<{ id: string; title: string }>('SELECT id, title FROM projects WHERE id = ? AND user_id = ? AND deleted_at IS NULL', [projectId, userId])
  1080→    if (projRows.length === 0) {
  1081→      await render(token, chatId, messageId, t(lang, 'Project not found.', 'پروژه پیدا نشد.'), homeKeyboard(lang))
  1082→      return
  1083→    }
  1084→    // Attach the note to the project (quick_notes.project_id — migration 0017). rule 1: user_id filter.
  1085→    await db.execute('UPDATE quick_notes SET project_id = ? WHERE id = ? AND user_id = ?', [projectId, session.noteId, userId])
  1086→    await clearSession(db, userId)
  1087→    await render(
  1088→      token,
  1089→      chatId,
  1090→      messageId,
  1091→      `🔗 ${t(lang, 'Connected to', 'متصل شد به')} <b>${esc(projRows[0].title)}</b>\n\n<a href="${origin}/project.html?id=${projectId}">${t(lang, 'Open in Hibana →', 'باز کردن در هیبانا →')}</a>`,
  1092→      kb([[{ text: t(lang, '🏠 Home', '🏠 خانه'), callback_data: 'home' }]]),
  1093→    )
  1094→    return
  1095→  }
  1096→  // 🔍 Search → await_text:proj_search (carry the noteId).
  1097→  if (data === 'psearch') {
  1098→    const session = await getSession(db, userId)
  1099→    if (!session || session.kind !== 'connect_list') {
  1100→      await render(token, chatId, messageId, homeText(lang), homeKeyboard(lang))
  1101→      return
  1102→    }
  1103→    await setSession(db, userId, { kind: 'await_text', intent: 'proj_search', noteId: session.noteId })
  1104→    await render(token, chatId, messageId, t(lang, '🔍 Type a few letters of the project title:', '🔍 چند حرف از عنوان پروژه را بنویس:'), cancelKeyboard(lang))
  1105→    return
  1106→  }
  1107→
  1108→  // Unknown callback — show home.
  1109→  await render(token, chatId, messageId, homeText(lang), homeKeyboard(lang))
  1110→}

  1114→// notes is a no-op: a title that already exists (case-insensitive, per user) is skipped.
  1115→export function registerImport(app: Hono<{ Variables: { user: UserRow } }>, cfg: Config) {
  1116→  app.post('/api/import/obsidian', requireAuth(cfg), async (c) => {
  1117→    const user = c.get('user')
  1118→    // Heavy-endpoint metering (hardening 2026-08-28): import inflates archives and writes
  1119→    // unbounded rows — 5 per minute per IP is generous for a real vault import.
  1120→    if (await hitRateLimit(cfg.db, RATE_RULES.import, clientIp(c))) {
  1121→      return c.json({ error: 'rate_limited', message: 'Too many imports — wait a minute and try again.' }, 429)
  1122→    }
  1123→    const form = await c.req.formData().catch(() => null)
  1124→    if (!form) return c.json({ error: 'invalid_input' }, 400)
  1125→
  1126→    const allFiles = (form.getAll('files') as File[]).filter((f) => f && typeof f.arrayBuffer === 'function')
  1127→    const mdFiles = allFiles.filter((f) => /\.md$/i.test(f.name ?? ''))
  1128→    const zipFiles = allFiles.filter((f) => /\.zip$/i.test(f.name ?? ''))
  1129→    if (mdFiles.length === 0 && zipFiles.length === 0) {
  1130→      return c.json({ error: 'no_markdown_files', message: 'Attach .md files or a .zip of your exported notes.' }, 400)
  1131→    }
  1132→    // Request-shape caps (hardening 2026-08-28): bounded file count + total upload bytes,
  1133→    // enforced before anything is read into memory. 100MB matches Cloudflare's body cap.
  1134→    const MAX_FILES = 200
  1135→    const MAX_UPLOAD_BYTES = 100 * 1024 * 1024
  1136→    if (allFiles.length > MAX_FILES) {
  1137→      return c.json({ error: 'too_many', message: `Attach at most ${MAX_FILES} files per import.` }, 400)
  1138→    }
  1139→    const totalBytes = allFiles.reduce((n, f) => n + (f.size ?? 0), 0)
  1140→    if (totalBytes > MAX_UPLOAD_BYTES) {
  1141→      return c.json({ error: 'too_many', message: 'Import is limited to 100MB of uploads per request.' }, 413)
  1142→    }
  1143→
  1144→    // rule 1: titles are scoped to this user; collect existing ones once for dedupe.
  1145→    const existing = await cfg.db.query<{ title: string }>("SELECT LOWER(title) AS title FROM projects WHERE user_id = ?", [user.id])
  1146→    const knownTitles = new Set(existing.map((r) => String(r.title).toLowerCase()))
  1147→
  1148→    const notes: Array<{ fileName: string; content: string }> = []
  1149→    for (const file of mdFiles) notes.push({ fileName: file.name, content: await file.text() })
  1150→    for (const file of zipFiles) {
  1151→      try {
  1152→        notes.push(...markdownFromZip(new Uint8Array(await file.arrayBuffer())))
  1153→      } catch (err) {
  1154→        if (err instanceof ZipLimitError) {
  1155→          return c.json({ error: 'too_many', message: err.message }, 413)
  1156→        }
  1157→        return c.json({ error: 'invalid_zip', message: 'Could not read that .zip file — is it a valid archive?' }, 400)
  1158→      }
  1159→    }
  1160→
  1161→    let count = 0
  1162→    let duplicates = 0
  1163→    // H1 fix (2026-09-10): resolve the 'Imported' tag id atomically BEFORE the transaction
  1164→    // loop. The old pattern queried inside the loop on every iteration — pre-batch reads see
  1165→    // stale state, so every iteration after the first tried to INSERT a duplicate 'Imported'
  1166→    // tag, hitting UNIQUE(user_id, name) and rolling back the ENTIRE import. A fresh-user
  1167→    // multi-note import was guaranteed to fail.
  1168→    const importedTagRows = await cfg.db.query<{ id: string }>(
  1169→      `INSERT INTO tags (id, user_id, name, color, usage_count, created_at)
  1170→       VALUES (?, ?, 'Imported', '#94a3b8', 0, ?)
  1171→       ON CONFLICT(user_id, name) DO UPDATE SET usage_count = tags.usage_count
  1172→       RETURNING id`,
  1173→      [uuid(), user.id, new Date().toISOString()],
  1174→    )
  1175→    const importedTagId = importedTagRows[0]?.id
  1176→    // P3.2 (F-M3): wrap the whole import loop in ONE transaction. Was ~200 sequential
  1177→    // iterations each doing an INSERT + a sub-transaction (3-4 round trips each). Now
  1178→    // batched: all INSERTs + tag links land atomically; a failure rolls back the whole
  1179→    // import (no half-imported state).
  1180→    await cfg.db.transaction(async (tx) => {
  1181→      for (const note of notes) {
  1182→        const parsed = parseMarkdownNote(note.fileName, note.content)
  1183→        if (!parsed) continue
  1184→        const key = parsed.title.toLowerCase()
  1185→        if (knownTitles.has(key)) {
  1186→          duplicates++
  1187→          continue
  1188→        }
  1189→        knownTitles.add(key)
  1190→
  1191→        const id = uuid()
  1192→        const now = new Date().toISOString()
  1193→        tx.sql(
  1194→          "INSERT INTO projects (id, user_id, title, description, type, status, sort_order, latest_note, reminders_enabled, created_at, updated_at) VALUES (?, ?, ?, ?, 'personal', 'spark', 0, '', 0, ?, ?)",
  1195→          [id, user.id, parsed.title, parsed.description, now, now],
  1196→        )
  1197→        if (importedTagId) tx.sql('INSERT OR IGNORE INTO project_tags (project_id, tag_id) VALUES (?, ?)', [id, importedTagId])
  1198→        count++
  1199→      }
  1200→    })
  1201→    return c.json({ ok: true, imported: count, duplicates })
  1202→  })
  1203→}
